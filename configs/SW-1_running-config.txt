version 15.2
service timestamps debug datetime msec
service timestamps log datetime msec
no service password-encryption

hostname SW-1

! --- VLANs ---

vlan 10
 name ADMINISTRATION

vlan 20
 name INFORMATIQUE

! --- Ports acces VLAN 10 (Fa0/1 a Fa0/12) ---

interface range FastEthernet0/1 - 12
 description Acces VLAN 10 - Administration
 switchport mode access
 switchport access vlan 10
 no shutdown

! --- Ports acces VLAN 20 (Fa0/13 a Fa0/24) ---

interface range FastEthernet0/13 - 24
 description Acces VLAN 20 - Informatique
 switchport mode access
 switchport access vlan 20
 no shutdown

! --- Port Trunk vers Routeur R4 ---

interface GigabitEthernet0/1
 description Trunk 802.1Q vers R4
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20
 no shutdown

! --- Interface de gestion SVI VLAN 10 ---

interface Vlan10
 description Interface de gestion Switch
 ip address 192.168.10.2 255.255.255.0
 no shutdown

! --- Passerelle par defaut ---

ip default-gateway 192.168.10.1

! --- SNMP ---

snmp-server community public RO
snmp-server location EVE-NG Lab - SUP de CO Dakar
snmp-server contact Aliou Barry - Admin Reseau
snmp-server host 192.168.10.50 version 2c public

! --- Fin de configuration ---
end
